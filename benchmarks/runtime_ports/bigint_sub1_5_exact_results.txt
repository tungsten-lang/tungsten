literal native Tungsten port of runtime.c's bn_sub_word_a64_fixed case 5:

  ldp x4, x5, [src]

  ldr x8, [src, #32]
  subs x4, x4, word
  sbcs x5, x5, xzr
  sbcs x6, x6, xzr
  sbcs x7, x7, xzr
  sbcs x8, x8, xzr
  stp x4, x5, [dst]
  stp x6, x7, [dst, #16]
  str x8, [dst, #32]
  cset borrow, lo

same order as bigint_sub_ui_any.  No borrow-death, store scheduling,
allocator specialization, direct dispatch seam, or representation change is
included.
